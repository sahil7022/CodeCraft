code stack.cpp 
#include <iostream>
#include <vector>

class Stack {
private:
    std::vector<int> elements;
    
public:
    // PUSH OPERATION
    void push(int val) {

        elements.push_back(val);
        std::cout << "Pushed " << val << std::endl;
    }

    // POP OPERATION
    void pop() {

        if (elements.empty()) return;
        elements.pop_back();
    }
    
    // PEEK OPERATION
    int peek() {

        if (elements.empty()) return -1;
        return elements.back();
    }
};
speed TIME COMPLEXITY
Push() O(1)
Pop() O(1)
